```python
import secrets
import os
from datetime import datetime, timedelta
from typing import Optional, Tuple
from sqlmodel import select
from sqlmodel.ext.asyncio.session import AsyncSession
from pwdlib import PasswordHash
from src.db.models import OTPCode
import resend

# Configure
OTP_EXPIRE_MINUTES = int(os.environ.get("OTP_EXPIRE_MINUTES", "10"))
OTP_MAX_REQUESTS_PER_HOUR = 5
OTP_MAX_ATTEMPTS_PER_CODE = 5
RESEND_API_KEY = os.environ.get("RESEND_API_KEY", "")
RESEND_FROM_EMAIL = os.environ.get("RESEND_FROM_EMAIL", "noreply@yourdomain.com")

hasher = PasswordHash.recommended()

class OTPService:
    @staticmethod
    def generate_code() -> str:
        """Generate 6-digit OTP code."""
        return f"{secrets.randbelow(1000000):06d}"

    @staticmethod
    async def check_rate_limit(db: AsyncSession, email: str) -> Tuple[bool, str]:
        """Check if email has exceeded rate limits. Returns (allowed, error_message)."""
        one_hour_ago = datetime.utcnow() - timedelta(hours=1)

        # Count OTP requests in last hour
        stmt = select(OTPCode).where(
            OTPCode.email == email,
            OTPCode.created_at > one_hour_ago
        )
        result = await db.exec(stmt)
        recent_codes = result.all()

        if len(recent_codes) >= OTP_MAX_REQUESTS_PER_HOUR:
            return False, "too_many_requests"

        return True, ""

    @staticmethod
    async def invalidate_previous_codes(db: AsyncSession, email: str) -> None:
        """Mark all previous codes for this email as used (invalidate them)."""
        stmt = select(OTPCode).where(
            OTPCode.email == email,
            OTPCode.used_at.is_(None)
        )
        result = await db.exec(stmt)
        for code in result.all():
            code.used_at = datetime.utcnow()
        await db.commit()

    @staticmethod
    async def create_code(db: AsyncSession, email: str) -> str:
        """Create and store new OTP code. Returns the plaintext code."""
        # Invalidate previous codes (AUTH-03)
        await OTPService.invalidate_previous_codes(db, email)

        code = OTPService.generate_code()
        code_hash = hasher.hash(code)
        expires_at = datetime.utcnow() + timedelta(minutes=OTP_EXPIRE_MINUTES)

        otp = OTPCode(
            email=email,
            code_hash=code_hash,
            expires_at=expires_at,
            attempts=0
        )
        db.add(otp)
        await db.commit()

        return code

    @staticmethod
    async def send_otp_email(email: str, code: str) -> bool:
        """Send OTP code via email. Returns success status."""
        if not RESEND_API_KEY:
            # Dev mode: log code instead of sending
            print(f"[DEV] OTP for {email}: {code}")
            return True

        resend.api_key = RESEND_API_KEY
        try:
            resend.Emails.send({
                "from": RESEND_FROM_EMAIL,
                "to": email,
                "subject": "Your verification code",
                "html": f"""
                <h2>Your verification code</h2>
                <p style="font-size: 32px; font-weight: bold; letter-spacing: 8px;">{code}</p>
                <p>This code expires in {OTP_EXPIRE_MINUTES} minutes.</p>
                <p>If you didn't request this code, you can safely ignore this email.</p>
                """
            })
            return True
        except Exception as e:
            print(f"Failed to send OTP email: {e}")
            return False

    @staticmethod
    async def verify_code(
        db: AsyncSession,
        email: str,
        code: str
    ) -> Tuple[bool, str, Optional[OTPCode]]:
        """
        Verify OTP code. Returns (success, error_code, otp_record).
        Error codes: 'invalid', 'expired', 'too_many_attempts', 'already_used'
        """
        stmt = select(OTPCode).where(
            OTPCode.email == email,
            OTPCode.used_at.is_(None)
        ).order_by(OTPCode.created_at.desc())

        result = await db.exec(stmt)
        otp = result.first()

        if not otp:
            return False, "invalid", None

        # Check attempts (AUTH-05)
        if otp.attempts >= OTP_MAX_ATTEMPTS_PER_CODE:
            return False, "too_many_attempts", None

        # Increment attempts
        otp.attempts += 1
        await db.commit()

        # Check expiration (AUTH-02)
        if datetime.utcnow() > otp.expires_at:
            return False, "expired", None

        # Verify code hash
        if not hasher.verify(code, otp.code_hash):
            return False, "invalid", None

        # Mark as used
        otp.used_at = datetime.utcnow()
        await db.commit()

        return True, "", otp
```

from fastapi import APIRouter, Depends, HTTPException, Request
from pydantic import BaseModel, EmailStr
from sqlmodel.ext.asyncio.session import AsyncSession
from src.dependencies.database import get_db
from src.services.otp import OTPService

auth_router = APIRouter(prefix="/auth", tags=["auth"])

class OTPSendRequest(BaseModel):
    email: EmailStr

class OTPSendResponse(BaseModel):
    success: bool
    message: str

class OTPVerifyRequest(BaseModel):
    email: EmailStr
    code: str

class OTPVerifyResponse(BaseModel):
    success: bool
    error: str | None = None
    # user and workspace added in Plan 02-02

@auth_router.post("/otp/send", response_model=OTPSendResponse)
async def send_otp(
    request: OTPSendRequest,
    db: AsyncSession = Depends(get_db)
):
    """
    Send OTP code to email address.

    Rate limited to 5 requests per email per hour (AUTH-04).
    """
    # Check rate limit
    allowed, error = await OTPService.check_rate_limit(db, request.email)
    if not allowed:
        raise HTTPException(
            status_code=429,
            detail={
                "error": error,
                "message": "Too many OTP requests. Please try again later."
            }
        )

    # Create and send OTP
    code = await OTPService.create_code(db, request.email)
    sent = await OTPService.send_otp_email(request.email, code)

    if not sent:
        raise HTTPException(
            status_code=500,
            detail={"error": "email_failed", "message": "Failed to send verification email"}
        )

    return OTPSendResponse(
        success=True,
        message="Verification code sent to your email"
    )

@auth_router.post("/otp/verify", response_model=OTPVerifyResponse)
async def verify_otp(
    request: OTPVerifyRequest,
    client_request: Request,
    db: AsyncSession = Depends(get_db)
):
    """
    Verify OTP code.

    Returns specific error messages for different failure cases (AUTH-06):
    - 'invalid': Code doesn't match
    - 'expired': Code has expired
    - 'too_many_attempts': Max verification attempts exceeded
    """
    success, error_code, otp_record = await OTPService.verify_code(
        db, request.email, request.code
    )

    if not success:
        # Map error codes to user-friendly messages (AUTH-06)
        error_messages = {
            "invalid": "Invalid verification code",
            "expired": "Verification code has expired. Please request a new one.",
            "too_many_attempts": "Too many verification attempts. Please request a new code.",
            "already_used": "This code has already been used"
        }
        raise HTTPException(
            status_code=400,
            detail={
                "error": error_code,
                "message": error_messages.get(error_code, "Verification failed")
            }
        )

    # Successful verification - user/workspace creation handled in Plan 02-02
    return OTPVerifyResponse(success=True)
