```python
"""
Final summary LLM pass for the v2.6 cost-driver-first pipeline.

Aggregates all per-driver DriverAnalysisResult objects into a single
executive overview via one dedicated LLM request.

Key design constraints (Phase 32 requirements):
- SUMM-01: Context contains aggregated driver narratives — no raw line items
- SUMM-02: generate_structured() only — no JSON repair fallback
- SUMM-01: quality_notes parameter enables single-pass rewrite without separate prompt

No circular import issue: summary_pass.py imports only from pipeline.models and
pipeline.cache — no bid_comp dependency.
"""
from __future__ import annotations

import json
import logging
from typing import Any, Dict, List, Optional

from pydantic import BaseModel, Field

from vip_shared.pipeline.cache import PipelineCache, cache_key
from vip_shared.pipeline.models import DriverAnalysisResult, SummaryResult
from vip_shared.llm.adapter import LLMAdapterBase


logger = logging.getLogger("vip-parse.pipeline.summary_pass")


class SummaryPassInput(BaseModel):
    """
    Pydantic model used to compute the content-hash cache key for a summary pass.

    quality_notes is included so a rewrite call (non-empty notes) has a different
    cache key from the initial call (empty notes) — rewrites are never served from
    the initial pass cache.
    """

    primary_name: str = Field(description="Primary estimate name")
    comparison_name: str = Field(description="Comparison estimate name")
    driver_summaries_json: str = Field(description="JSON string of aggregated driver summaries")
    quality_notes: str = Field(default="", description="Quality failure details for rewrite path")


def run_summary_pass(
    driver_analyses: List[DriverAnalysisResult],
    llm_adapter: LLMAdapterBase,
    primary_name: str = "Primary Estimate",
    comparison_name: str = "Comparison Estimate",
    cache: Optional[PipelineCache] = None,
    quality_notes: str = "",
) -> SummaryResult:
    """
    Run the final summary LLM pass — aggregate all driver analyses into one executive overview.

    SUMM-01: Context contains aggregated driver summaries (narratives, deltas, observations).
             Raw line items are NOT included — those were consumed by run_driver_pass().
    SUMM-02: generate_structured() is the only LLM call path — no JSON repair fallback.
             If generate_structured raises, the exception propagates to the caller.
    Rewrite: When quality_notes is non-empty, it is included in context so the LLM
             revises its output. The CostDriverPipeline calls this once normally, then
             once with quality_notes if GATE-01 or GATE-02 fails (max 1 rewrite).

    Args:
        driver_analyses: List of DriverAnalysisResult from run_driver_pass() calls.
        llm_adapter: LLM adapter implementing generate_structured().
        primary_name: Name of the primary estimate (for context).
        comparison_name: Name of the comparison estimate (for context).
        cache: Optional PipelineCache. If None, caching is skipped.
        quality_notes: Empty string for initial pass; failure details for rewrite path.

    Returns:
        SummaryResult with overview, scope_observations, and suggested_followups.

    Raises:
        Any exception from generate_structured() — no fallback (SUMM-02).
    """
    # --- Cache: try first ---
    key: Optional[str] = None
    if cache is not None:
        driver_summaries = _build_driver_summaries(driver_analyses)
        driver_summaries_json = json.dumps(driver_summaries, indent=2)
        pass_input = SummaryPassInput(
            primary_name=primary_name,
            comparison_name=comparison_name,
            driver_summaries_json=driver_summaries_json,
            quality_notes=quality_notes,
        )
        key = cache_key("summary_pass", pass_input)
        cached = cache.get(key, SummaryResult)
        if cached is not None:
            logger.info(
                "summary_pass cache hit: drivers=%d key=%s",
                len(driver_analyses),
                key[:40],
            )
            return cached

    # --- Build context (aggregated summaries, not raw items) ---
    driver_summaries = _build_driver_summaries(driver_analyses)
    context: Dict[str, Any] = {
        "primary_name": primary_name,
        "comparison_name": comparison_name,
        "driver_count": len(driver_analyses),
        "driver_summaries_json": json.dumps(driver_summaries, indent=2),
        "quality_notes": quality_notes,
    }

    logger.info(
        "summary_pass start: drivers=%d primary=%s comparison=%s rewrite=%s",
        len(driver_analyses),
        primary_name,
        comparison_name,
        "yes" if quality_notes else "no",
    )

    # --- generate_structured only — no fallback (SUMM-02) ---
    result: SummaryResult = llm_adapter.generate_structured(
        "final_summary_v1",
        context,
        response_model=SummaryResult,
    )

    logger.info(
        "summary_pass complete: overview_chars=%d observations=%d followups=%d",
        len(result.overview),
        len(result.scope_observations),
        len(result.suggested_followups),
    )

    # --- Cache: store on miss ---
    if cache is not None and key is not None:
        cache.set(key, result)

    return result


def _build_driver_summaries(driver_analyses: List[DriverAnalysisResult]) -> list:
    """Build aggregated driver summaries for LLM context (no raw line items)."""
    return [
        {
            "category": d.category,
            "primary_total": d.primary_total,
            "comparison_total": d.comparison_total,
            "delta": d.delta,
            "narrative": d.narrative,
            "scope_observations": d.scope_observations,
            "suggested_followups": d.suggested_followups,
        }
        for d in driver_analyses
    ]
```
