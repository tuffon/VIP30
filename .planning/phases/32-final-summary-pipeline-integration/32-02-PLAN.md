"""
CostDriverPipeline — v2.6 cost-driver-first orchestrator.

Replaces NarrativePipeline with a stepped architecture:
1. build_trade_context() — normalized category totals from parser recap
2. identify_cost_drivers() — top-N categories by absolute delta
3. map_driver_items() — all line items per driver category
4. run_driver_pass() × N — per-driver LLM request (isolated context)
5. run_summary_pass() — aggregate all driver analyses into executive overview
6. Quality check (GATE-01/02 only) — optional single rewrite pass

Design constraints (Phase 32 requirements):
- REWRITE-01: Quality rewrite triggers ONLY on GATE-01 (HedgingChecker) or
  GATE-02 (JudgmentLanguageChecker) failure. Max 1 rewrite attempt. No loop.
- REWRITE-02: No _finalize_with_error or placeholder text. run_summary_pass
  exceptions propagate to BidComp._generate_narrative_via_pipeline() which
  already handles them with _fallback_narrative().
- REWRITE-03: Failed run_driver_pass() → explicit DriverNarrative with
  "Analysis unavailable. Delta: $X.XX" text. Pipeline continues with remaining
  drivers and still calls run_summary_pass with the successful analyses.

Interface contract:
  CostDriverPipeline.run() has the same signature as NarrativePipeline.run()
  so BidComp._generate_narrative_via_pipeline() requires no changes.
"""
from __future__ import annotations

import logging
import time
from typing import Any, Dict, List, Optional

from vip_shared.llm.adapter import LLMAdapterBase
from vip_shared.methodology.models import MethodologyResult
from vip_shared.rules.models import SignalBundle

from .cache import PipelineCache
from .models import (
    DriverNarrative,
    FinalNarrative,
    QualityCheckResult,
    QualityReport,
    SummaryResult,
)
from .passes import (
    build_trade_context,
    identify_cost_drivers,
    map_driver_items,
    run_driver_pass,
    run_summary_pass,
)
from .quality import HedgingChecker, JudgmentLanguageChecker
from .state import PipelineState


logger = logging.getLogger("vip-parse.pipeline.cost_driver")


class CostDriverPipeline:
    """
    Cost-driver-first pipeline orchestrator (v2.6).

    Each top cost driver gets its own LLM request with isolated context.
    A final summary pass aggregates all driver analyses into an executive overview.

    Usage:
        pipeline = CostDriverPipeline(llm_adapter)
        state = pipeline.run(pair, top_deltas, primary_name, comparison_name)
        final_narrative = state.final

    With caching:
        cache = PipelineCache(redis_client)
        pipeline = CostDriverPipeline(llm_adapter, cache=cache)
    """

    TOP_N_DRIVERS = 5

    def __init__(
        self,
        llm_adapter: LLMAdapterBase,
        cache: Optional[PipelineCache] = None,
    ):
        self.llm_adapter = llm_adapter
        self.cache = cache
        self.logger = logging.getLogger("vip-parse.pipeline.cost_driver")

    def run(
        self,
        pair: Any,  # EstimatePair — use Any to avoid circular import
        top_deltas: List[Dict[str, Any]],
        primary_name: str,
        comparison_name: str,
        methodology: Optional[MethodologyResult] = None,
        signals: Optional[SignalBundle] = None,
    ) -> PipelineState:
        """
        Execute the full cost-driver pipeline and return PipelineState.

        Args:
            pair: EstimatePair with primary and comparison estimates
            top_deltas: Category deltas from BidComp (not used for driver identification
                        — TradeContext is built from pair payloads directly)
            primary_name: Name of the primary estimate
            comparison_name: Name of the comparison estimate
            methodology: Optional pre-LLM methodology comparison
            signals: Optional pre-LLM intelligence signals

        Returns:
            PipelineState with state.final (FinalNarrative) and state.quality_report set.

        Raises:
            Any exception from run_summary_pass() — REWRITE-02: no _finalize_with_error.
        """
        state = PipelineState(
            pair=pair,
            top_deltas=top_deltas,
            methodology=methodology,
            signals=signals,
        )

        self.logger.info(
            "cost_driver_pipeline start: primary=%s comparison=%s",
            primary_name,
            comparison_name,
        )

        # Step 1: Build trade context from parser recap data
        start_ms = _now_ms()
        trade_context = build_trade_context(pair.primary.payload, pair.comparison.payload)
        state.add_timing("trade_context", _now_ms() - start_ms)
        state.mark_pass_executed("trade_context")

        # Step 2: Identify top cost drivers by absolute dollar delta
        start_ms = _now_ms()
        drivers = identify_cost_drivers(trade_context, top_n=self.TOP_N_DRIVERS)
        state.add_timing("identify_drivers", _now_ms() - start_ms)

        # Step 3: Map line items for each driver
        start_ms = _now_ms()
        drivers_with_items = map_driver_items(drivers, pair.primary.payload, pair.comparison.payload)
        state.add_timing("map_driver_items", _now_ms() - start_ms)

        self.logger.info(
            "cost_driver_pipeline: drivers=%d items_mapped=%d",
            len(drivers),
            len(drivers_with_items),
        )

        # Step 4: Per-driver LLM pass — REWRITE-03: failed driver → unavailable narrative
        driver_analyses = []
        key_drivers = []

        for dwi in drivers_with_items:
            start_ms = _now_ms()
            try:
                analysis = run_driver_pass(dwi, self.llm_adapter, cache=self.cache)
                driver_analyses.append(analysis)
                key_drivers.append(
                    DriverNarrative(category=dwi.driver.category, narrative=analysis.narrative)
                )
                state.add_timing(f"driver_pass:{dwi.driver.category}", _now_ms() - start_ms)
                state.mark_pass_executed(f"driver_pass:{dwi.driver.category}")
                self.logger.info(
                    "driver_pass complete: category=%s narrative_chars=%d",
                    dwi.driver.category,
                    len(analysis.narrative),
                )
            except Exception as exc:
                # REWRITE-03: explicit "analysis unavailable" entry — no silent fallback
                state.add_timing(f"driver_pass:{dwi.driver.category}", _now_ms() - start_ms)
                state.errors.append({
                    "pass": f"driver_pass:{dwi.driver.category}",
                    "error": str(exc),
                    "error_type": type(exc).__name__,
                })
                self.logger.warning(
                    "driver_pass failed: category=%s error=%s", dwi.driver.category, exc
                )
                key_drivers.append(DriverNarrative(
                    category=dwi.driver.category,
                    narrative=f"Analysis unavailable. Delta: ${dwi.driver.delta:+,.2f}",
                ))

        # Step 5: Final summary LLM pass — REWRITE-02: exception propagates
        start_ms = _now_ms()
        summary: SummaryResult = run_summary_pass(
            driver_analyses,
            self.llm_adapter,
            primary_name=primary_name,
            comparison_name=comparison_name,
            cache=self.cache,
        )
        state.add_timing("summary_pass", _now_ms() - start_ms)
        state.mark_pass_executed("summary_pass")

        # Step 6: Quality check (GATE-01 and GATE-02 only — REWRITE-01)
        rewrites = 0
        quality_notes, quality_report = self._check_quality(summary.overview)

        if quality_notes:
            # Single rewrite — no loop (REWRITE-01: max 1 attempt)
            self.logger.info("quality gate failed; requesting rewrite: %s", quality_notes[:80])
            start_ms = _now_ms()
            summary = run_summary_pass(
                driver_analyses,
                self.llm_adapter,
                primary_name=primary_name,
                comparison_name=comparison_name,
                cache=self.cache,
                quality_notes=quality_notes,
            )
            state.add_timing("summary_rewrite", _now_ms() - start_ms)
            state.mark_pass_executed("summary_rewrite")
            rewrites = 1
            # Re-check quality after rewrite (for quality_report only; no second rewrite)
            _, quality_report = self._check_quality(summary.overview)

        # Step 7: Assemble FinalNarrative (INTEG-02)
        state.quality_report = quality_report
        state.final = FinalNarrative(
            overview=summary.overview,
            key_drivers=key_drivers,
            scope_observations=summary.scope_observations,
            suggested_followups=summary.suggested_followups,
            quality_report=quality_report,
            rewrites_performed=rewrites,
        )

        self.logger.info(
            "cost_driver_pipeline complete: drivers=%d rewrites=%d quality_passed=%s",
            len(driver_analyses),
            rewrites,
            state.quality_passed(),
        )

        return state

    def _check_quality(self, overview: str) -> tuple[str, QualityReport]:
        """
        Run GATE-01 (HedgingChecker) and GATE-02 (JudgmentLanguageChecker) on overview text.

        Returns:
            (quality_notes, QualityReport) — quality_notes is empty string when both pass.
        """
        hedging_result = HedgingChecker().check(overview)
        judgment_result = JudgmentLanguageChecker().check(overview)

        checks = [
            QualityCheckResult(
                check_name="hedging",
                passed=hedging_result.passed,
                details=hedging_result.details if not hedging_result.passed else "",
            ),
            QualityCheckResult(
                check_name="judgment_language",
                passed=judgment_result.passed,
                details=judgment_result.details if not judgment_result.passed else "",
            ),
        ]
        all_passed = hedging_result.passed and judgment_result.passed
        report = QualityReport(passed=all_passed, checks=checks)

        if all_passed:
            return "", report

        # Build quality_notes string for rewrite prompt
        failed_details = []
        if not hedging_result.passed:
            failed_details.append(f"GATE-01 (hedging): {hedging_result.details}")
        if not judgment_result.passed:
            failed_details.append(f"GATE-02 (judgment): {judgment_result.details}")
        quality_notes = "\n".join(failed_details)

        return quality_notes, report
