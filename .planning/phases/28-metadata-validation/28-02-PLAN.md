```bash
    ('bschacter',    'docs/final-drafts/BSchacter-02.12.26-Est-JVB-RepairEstimate-\$809,464.83.pdf', GOLDEN_DIR / 'bschacter.golden.json'),
    ('SF_BSchacter', 'docs/final-drafts/statefarm/SF_BSchacter.pdf',                                  GOLDEN_DIR / 'statefarm/SF_BSchacter.golden.json'),
    ('lachman_sf',   'docs/final-drafts/statefarm/Estimate SF Structural damage Lachman 4.15.2025.pdf', GOLDEN_DIR / 'statefarm/lachman_sf.golden.json'),
    ('kalyvas_sf',   'docs/final-drafts/statefarm/Kalyvas Preliminary State Farm estimate9-25-25.pdf',  GOLDEN_DIR / 'statefarm/kalyvas_sf.golden.json'),
]
META_FIELDS = ['insured_name', 'price_list', 'property_address', 'claim_number']

results = {}
for name, pdf_path, golden_path in DOCS:
    print(f'Parsing {name}...')
    with tempfile.TemporaryDirectory() as tmpdir:
        parser = XactimateRoughDraftParser(pdf_path, tmpdir)
        parser.run()
        stem = Path(pdf_path).stem
        new_data = json.loads((Path(tmpdir) / f'{stem}.json').read_text(encoding='utf-8'))
    with open(golden_path, encoding='utf-8') as f:
        old_data = json.load(f)
    new_secs = new_data.get('sections', [])
    old_secs = old_data.get('sections', [])
    new_items = sum(len(s.get('line_items', []) or []) for s in new_secs)
    old_items = sum(len(s.get('line_items', []) or []) for s in old_secs)
    new_meta = new_data.get('case_metadata') or {}
    old_meta = old_data.get('case_metadata') or {}
    meta_changes = []
    for f in META_FIELDS:
        ov, nv = old_meta.get(f), new_meta.get(f)
        if ov != nv:
            meta_changes.append(f'    {f}: {ov!r} -> {nv!r}')
    results[name] = {'new_data': new_data, 'old_secs': len(old_secs), 'new_secs': len(new_secs), 'old_items': old_items, 'new_items': new_items, 'meta_changes': meta_changes}
    print(f'  Sections: {len(old_secs)} -> {len(new_secs)}  Items: {old_items} -> {new_items}')
    if meta_changes:
        print('  Metadata changes:')
        for c in meta_changes: print(c)
    else:
        print('  Metadata: no changes')
    print()
" 2>&1
```
